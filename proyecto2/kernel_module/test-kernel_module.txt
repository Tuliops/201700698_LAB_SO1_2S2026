# Compilar el modulo de kernel
make

# Compilar e insertar el modulo en el sistema
make load

# Probar la lectura del archivo /proc
make test

# Desmontar el modulo del kernel
make unload

# Limpiar archivos de compilacion
make clean




#==========  COMANDOS MANUALES DE KERNEL ==========
# Insertar el modulo manualmente
sudo insmod continfo.ko

# Remover el modulo manualmente
sudo rmmod continfo

# Verificar si el modulo está cargado y activo
lsmod | grep continfo

# Inspeccionar información del modulo (autor, descripción, licencia)
modinfo continfo.ko

# Ver mensajes del kernel impresos por pr_info() y pr_err()
sudo dmesg | tail -n 20

# Limpiar el buffer de logs de dmesg para hacer pruebas limpias
sudo dmesg -c


#### ==========  VALIDACIONES Y PURBAS DEL ENDPINT ================
# Leer directamente el contenido del archivo /proc
cat /proc/continfo_pr2_so1_201700698

# Validar que la sintaxis JSON sea correcta y ver los primeros procesos
cat /proc/continfo_pr2_so1_201700698 | jq . | head -n 30

# Filtrar procesos específicos por nombre (ejemplo: buscar 'bash' o 'systemd')
cat /proc/continfo_pr2_so1_201700698 | jq '.[] | select(.name=="bash")'

# Contar cuántos procesos está detectando el modulo en tiempo real
cat /proc/continfo_pr2_so1_201700698 | jq '. | length'
